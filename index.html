async function sharePDF() { if (!currentBuyerData) { return; } try { pdfButton.disabled = true; pdfButton.textContent = "Creating PDF..."; const { jsPDF } = window.jspdf; const data = currentBuyerData; const doc = new jsPDF({ orientation: "portrait", unit: "mm", format: "a4" }); const buyerName = data.buyer || "Buyer"; const fileName = buyerName + " Outstanding List.pdf"; // ==================================================== // PAGE / LAYOUT CONSTANTS // ==================================================== const pageWidth = 210; const leftMargin = 10; const tableWidth = 190; // ==================================================== // TITLE // ==================================================== doc.setFont( "helvetica", "bold" ); doc.setFontSize(16); doc.setTextColor( 0, 0, 0 ); doc.text( "Outstanding List", pageWidth / 2, 12, { align: "center" } ); // ==================================================== // BUYER NAME // // Keep our current Buyer Name style: // BLUE BACKGROUND + WHITE TEXT // ==================================================== doc.setFillColor( 173, 216, 230 ); doc.rect( 27.5, 16, 155, 6, "F" ); doc.setFont( "helvetica", "bold" ); doc.setFontSize(11); doc.setTextColor( 255, 255, 255 ); doc.text( buyerName, pageWidth / 2, 20.3, { align: "center" } ); // ==================================================== // DATE + PK AGENCY // // Same placement as reference. // ==================================================== const today = new Date() .toLocaleDateString( "en-GB" ); doc.setTextColor( 0, 0, 0 ); doc.setFont( "helvetica", "bold" ); doc.setFontSize(9); // Date doc.text( today, 55, 30, { align: "center" } ); // PK Agency doc.text( "PK AGENCY - 9425436227, 9406600197", 150, 30, { align: "center" } ); // ==================================================== // BILL TABLE DATA // ==================================================== const billRows = (data.bills || []).map( function (bill, index) { return [ String(index + 1), bill["Mumbai Party"] || "", String( bill["Bill Number"] || "" ), bill["Bill Date"] || "", formatIndianNumber( bill.Amount ), String( bill["Days Due"] || 0 ) ]; } ); const total = Number( data.total_outstanding || 0 ); // ==================================================== // BILL TABLE // ==================================================== doc.autoTable({ startY: 34, margin: { left: leftMargin, right: leftMargin }, tableWidth: tableWidth, head: [[ "S.No", "Supplier", "Bill Number", "Bill Date", "Amount", "Days Due" ]], body: billRows, foot: [[ "", "", "", "Grand Total", formatIndianNumber(total), "" ]], theme: "grid", styles: { font: "helvetica", fontSize: 7.5, cellPadding: { top: 1.5, bottom: 1.5, left: 2, right: 2 }, textColor: [ 0, 0, 0 ], lineColor: [ 170, 170, 170 ], lineWidth: 0.15, valign: "middle" }, headStyles: { fillColor: [ 255, 255, 255 ], textColor: [ 0, 0, 0 ], fontStyle: "bold", fontSize: 8, halign: "center", lineColor: [ 170, 170, 170 ], lineWidth: 0.15 }, footStyles: { fillColor: [ 232, 232, 232 ], textColor: [ 0, 0, 0 ], fontStyle: "bold", fontSize: 8 }, columnStyles: { // S.No 0: { cellWidth: 14, halign: "center" }, // Supplier 1: { cellWidth: 66, halign: "left" }, // Bill Number 2: { cellWidth: 31, halign: "left" }, // Bill Date 3: { cellWidth: 30, halign: "left" }, // Amount 4: { cellWidth: 29, halign: "right" }, // Days Due 5: { cellWidth: 20, halign: "right" } }, // ================================================= // RED TEXT FOR BILLS > 120 DAYS // // Exactly like reference. // ================================================= didParseCell: function (hookData) { if ( hookData.section === "body" && hookData.column.index !== undefined ) { const rowIndex = hookData.row.index; if ( data.bills[rowIndex] ) { const days = Number( data.bills[rowIndex][ "Days Due" ] || 0 ); if (days > 120) { hookData.cell.styles.textColor = [ 255, 0, 0 ]; } } } }, // ================================================= // HEADER / FOOTER // ================================================= didDrawPage: function () { // Nothing here deliberately. // This prevents the title/date from // being duplicated on continuation pages. } }); // ==================================================== // SUPPLIER SUMMARY // ==================================================== let finalY = doc.lastAutoTable.finalY + 9; if (finalY > 265) { doc.addPage(); finalY = 15; } // ==================================================== // SUPPLIER SUMMARY TITLE // ==================================================== doc.setFont( "helvetica", "bold" ); doc.setFontSize(11); doc.setTextColor( 0, 0, 0 ); doc.text( "Supplier Wise Summary", leftMargin, finalY ); // ==================================================== // SUPPLIER SUMMARY DATA // ==================================================== const supplierRows = (data.supplier_summary || []).map( function (item) { return [ item.supplier || "", formatIndianNumber( item.total ) ]; } ); supplierRows.push([ "Grand Total", formatIndianNumber(total) ]); // ==================================================== // SUPPLIER SUMMARY TABLE // ==================================================== doc.autoTable({ startY: finalY + 4, margin: { left: leftMargin, right: leftMargin }, tableWidth: 170, head: [[ "Supplier", "Total Amount" ]], body: supplierRows, theme: "grid", styles: { font: "helvetica", fontSize: 8, cellPadding: { top: 1.8, bottom: 1.8, left: 2, right: 2 }, textColor: [ 0, 0, 0 ], lineColor: [ 170, 170, 170 ], lineWidth: 0.15 }, headStyles: { fillColor: [ 255, 255, 255 ], textColor: [ 0, 0, 0 ], fontStyle: "bold", halign: "center" }, columnStyles: { 0: { cellWidth: 110, halign: "left" }, 1: { cellWidth: 60, halign: "right" } }, didParseCell: function (hookData) { if ( hookData.section === "body" && hookData.row.index === supplierRows.length - 1 ) { hookData.cell.styles.fillColor = [ 232, 232, 232 ]; hookData.cell.styles.fontStyle = "bold"; } } }); // ==================================================== // CREATE PDF BLOB // ==================================================== const pdfBlob = doc.output("blob"); const pdfFile = new File( [pdfBlob], fileName, { type: "application/pdf" } ); // ==================================================== // SHARE ON MOBILE // ==================================================== if ( navigator.share && navigator.canShare && navigator.canShare({ files: [pdfFile] }) ) { await navigator.share({ title: buyerName + " Outstanding List", text: buyerName + " Outstanding List", files: [ pdfFile ] }); setStatus( "PDF shared successfully" ); } else { // ================================================= // DESKTOP / FALLBACK DOWNLOAD // ================================================= const url = URL.createObjectURL( pdfBlob ); const link = document.createElement( "a" ); link.href = url; link.download = fileName; document.body.appendChild( link ); link.click(); link.remove(); URL.revokeObjectURL( url ); setStatus( "PDF downloaded as: " + fileName ); } } catch (error) { console.error(error); if ( error.name === "AbortError" ) { setStatus( "PDF sharing cancelled" ); } else { setStatus( "PDF error: " + error.message, true ); } } finally { pdfButton.disabled = false; pdfButton.textContent = "Share PDF"; } }
